import java.util.*;

class Q1 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int A=0, B=0, C=0, D=0, F=0, total=0;

        for (int i=0; i<n; i++) {
            int m = sc.nextInt();
            total += m;

            if (m >= 90) { System.out.print("A "); A++; }
            else if (m >= 80) { System.out.print("B "); B++; }
            else if (m >= 70) { System.out.print("C "); C++; }
            else if (m >= 60) { System.out.print("D "); D++; }
            else { System.out.print("F "); F++; }
        }

        System.out.println("\nA=" + A + ", B=" + B + ", C=" + C +
                           ", D=" + D + ", F=" + F);
        System.out.println("Average = " + (double)total/n);
    }
}
