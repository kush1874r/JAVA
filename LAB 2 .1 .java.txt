import java.util.Scanner;

public class CustomerFeedback {

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter Customer Name: ");
        String customerName = sc.nextLine();

        System.out.print("Enter Feedback: ");
        String feedback = sc.nextLine();

        System.out.print("Enter Keyword: ");
        String keyword = sc.nextLine();

        if (feedback.trim().isEmpty()) {
            System.out.println("Invalid Feedback Message");
        }
        else if (feedback.length() > 500) {
            System.out.println("Feedback Exceeds Maximum Length");
        }
        else {

            // Count characters
            int characters = feedback.length();

            // Count words
            String[] words = feedback.trim().split("\\s+");
            int wordCount = words.length;

            // Convert to lowercase for case-insensitive search
            String lowerFeedback = feedback.toLowerCase();
            String lowerKeyword = keyword.toLowerCase();

            // Count keyword occurrences
            int count = 0;
            int position = 0;

            while ((position = lowerFeedback.indexOf(lowerKeyword, position)) != -1) {

                count++;

                position = position + lowerKeyword.length();
            }

            // Display results
            System.out.println("Customer Name: " + customerName);
            System.out.println("Characters = " + characters);
            System.out.println("Words = " + wordCount);

            if (count > 0) {
                System.out.println("Keyword Found = " + count);
            }
            else {
                System.out.println("Keyword Not Found");
            }
        }

        sc.close();
    }
}KU
